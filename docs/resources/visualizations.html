1. Kognitive Belastung: Verteilung im Arbeitsgedaechtnis

Die Arbeitsgedaechtniskapazitaet ist begrenzt. Entscheidend ist, wie die drei Arten kognitiver Belastung diesen begrenzten Raum aufteilen.

cognitive_load_data <- tibble(
  scenario = factor(
    rep(c("Schlechtes Design", "Gutes Design", "KI-Outsourcing"), each = 4),
    levels = c("Schlechtes Design", "Gutes Design", "KI-Outsourcing")
  ),
  load_type = factor(
    rep(c("Extrinsische Belastung", "Intrinsische Belastung",
          "Lernrelevante Verarbeitung", "Ungenutzte Kapazität"), 3),
    levels = c("Ungenutzte Kapazität", "Lernrelevante Verarbeitung",
               "Intrinsische Belastung", "Extrinsische Belastung")
  ),
  percentage = c(
    60, 30, 10, 0,
    10, 30, 60, 0,
    10, 30, 5, 55
  )
) |>
  filter(percentage > 0) |>
  group_by(scenario) |>
  mutate(
    y_pos = cumsum(percentage) - percentage / 2,
    label = if_else(percentage >= 8, paste0(percentage, "%"), "")
  ) |>
  ungroup()

load_colors <- c(
  "Intrinsische Belastung" = "#64748b",
  "Extrinsische Belastung" = "#dc2626",
  "Lernrelevante Verarbeitung" = "#16a34a",
  "Ungenutzte Kapazität" = "#e5e7eb"
)

ggplot(cognitive_load_data, aes(x = scenario, y = percentage, fill = load_type)) +
  geom_col(width = 0.7, color = "white", linewidth = 0.5) +
  geom_hline(yintercept = 100, linetype = "dashed", color = brand_dark, linewidth = 0.8) +
  geom_text(
    aes(y = y_pos, label = label),
    color = "white",
    size = 4,
    fontface = "bold",
    family = "Lato"
  ) +
  annotate(
    "text",
    x = 3.45, y = 102,
    label = "Kapazitätsgrenze",
    size = 3.5, hjust = 0, family = "Lato"
  ) +
  scale_fill_manual(values = load_colors, name = "Belastungsart") +
  scale_y_continuous(
    limits = c(0, 108),
    breaks = seq(0, 100, 25),
    labels = paste0(seq(0, 100, 25), "%")
  ) +
  labs(
    title = "Verteilung der kognitiven Belastung",
    subtitle = "Arbeitsgedächtniskapazität ist begrenzt\u2002\u2013\u2002die Verteilung macht den Unterschied",
    x = NULL,
    y = "Arbeitsgedächtniskapazität"
  ) +
  theme_workshop() +
  theme(
    axis.text.x = element_text(size = 11, face = "bold"),
    panel.grid.major.x = element_blank(),
    legend.position = "bottom",
    legend.title = element_text(face = "bold", size = 10),
    legend.text = element_text(size = 9),
    plot.margin = margin(10, 50, 10, 10)
  ) +
  guides(fill = guide_legend(nrow = 2, byrow = TRUE))

2. Expertise Reversal Effekt

Optimale Unterstuetzung haengt vom Vorwissen ab. Was Anfaengern hilft, schadet Experten, und umgekehrt. Permanentes KI-Scaffolding verhindert den Uebergang.

set.seed(42)

expertise_data <- tibble(
  expertise = seq(0, 10, 0.5)
) |>
  mutate(
    worked_examples = 85 - 7 * expertise,
    problem_solving = 35 + 4.5 * expertise,
    permanent_scaffolding = 40 + 0.5 * expertise
  ) |>
  pivot_longer(
    cols = c(worked_examples, problem_solving, permanent_scaffolding),
    names_to = "condition",
    values_to = "outcome"
  ) |>
  mutate(
    condition = factor(
      condition,
      levels = c("worked_examples", "problem_solving", "permanent_scaffolding"),
      labels = c(
        "Worked Examples / Hohe Anleitung",
        "Problemlösen / Niedrige Anleitung",
        "Permanentes KI-Scaffolding"
      )
    )
  )

crossing_x <- (85 - 35) / (7 + 4.5)
crossing_y <- 85 - 7 * crossing_x

ggplot(expertise_data, aes(x = expertise, y = outcome, color = condition, linetype = condition)) +
  geom_line(linewidth = 1.2) +
  geom_point(
    data = tibble(x = crossing_x, y = crossing_y),
    aes(x = x, y = y),
    color = brand_dark,
    size = 4, shape = 21, fill = "white", stroke = 1.5,
    inherit.aes = FALSE
  ) +
  annotate(
    "text",
    x = crossing_x + 0.3, y = crossing_y + 7,
    label = "Fading Point",
    size = 3.8, hjust = 0, fontface = "italic", family = "Lato"
  ) +
  annotate(
    "curve",
    x = crossing_x + 0.3, y = crossing_y + 5,
    xend = crossing_x + 0.05, yend = crossing_y + 1.5,
    curvature = 0.2,
    arrow = arrow(length = unit(0.15, "cm"), type = "closed"),
    color = "#666666"
  ) +
  scale_color_manual(
    values = c(
      "Worked Examples / Hohe Anleitung" = "#2563eb",

    ),
    name = NULL
  ) +
  scale_linetype_manual(
    values = c(
      "Worked Examples / Hohe Anleitung" = "solid",

    name = NULL
  ) +
  scale_x_continuous(
    breaks = c(0, 5, 10),

  scale_y_continuous(limits = c(20, 90), breaks = seq(20, 80, 20)) +
  labs(
    title = "Expertise Reversal Effekt",
    subtitle = "Optimale Unterstützung hängt vom Vorwissen ab",
    x = "Expertise-Level",
    y = "Lernergebnis"
  ) +
  theme_workshop() +
  theme(
    legend.position = "bottom",
    legend.text = element_text(size = 10),
    legend.key.width = unit(2, "cm")
  ) +
  guides(
    color = guide_legend(nrow = 2, byrow = TRUE),
    linetype = guide_legend(nrow = 2, byrow = TRUE)
  )

3. Testing Effekt

Aktives Abrufen fuehrt zu besserem Langzeitlernen als wiederholtes Studieren, selbst wenn die Gesamtlernzeit gleich bleibt.

testing_effect_data <- tibble(
  condition = factor(
    c("4\u00d7 Studieren", "3\u00d7 Studieren +\n1\u00d7 Test", "1\u00d7 Studieren +\n3\u00d7 Tests"),
    levels = c("4\u00d7 Studieren", "3\u00d7 Studieren +\n1\u00d7 Test", "1\u00d7 Studieren +\n3\u00d7 Tests")
  ),
  retention = c(40, 56, 61)
)

ggplot(testing_effect_data, aes(x = condition, y = retention)) +
  geom_col(
    aes(fill = retention),
    width = 0.65,
    color = brand_dark,
    linewidth = 0.3
  ) +
  geom_text(
    aes(label = paste0(retention, "%")),
    vjust = -0.5,
    size = 5,
    fontface = "bold",
    color = brand_dark,
    family = "Lato"
  ) +
  scale_fill_gradient(low = "#f0a5bb", high = brand_accent, guide = "none") +

  ) +

    title = "Testing Effekt",

  )

4. Das Lernparadox

Subjektives Gefuehl und tatsaechliches Lernen koennen auseinanderklaffen. Die metakognitive Falle: Was sich leicht anfuehlt, fuehlt sich nach Lernen an, ist es aber oft nicht.

quadrants <- tibble(
  xmin = c(0, 5, 0, 5),
  xmax = c(5, 10, 5, 10),
  ymin = c(5, 5, 0, 0),
  ymax = c(10, 10, 5, 5),
  quadrant = c(
    "Produktive\nAnstrengung",
    "Experten-\nEffizienz",
    "Unproduktive\nFrustration",
    "Illusion des\nLernens"
  ),
  fill_color = c("#16a34a", "#2563eb", "#9ca3af", "#dc2626"),
  importance = c("ZIEL", "OK für Fortgeschrittene", "Vermeiden", "FALLE")
)

ggplot(quadrants) +
  geom_rect(
    aes(xmin = xmin, xmax = xmax, ymin = ymin, ymax = ymax, fill = I(fill_color)),
    alpha = 0.3,
    color = brand_dark,
    linewidth = 0.8
  ) +
  geom_text(
    aes(
      x = (xmin + xmax) / 2,
      y = (ymin + ymax) / 2 + 0.7,
      label = quadrant
    ),
    size = 5.5,
    fontface = "bold",
    lineheight = 0.9,
    color = brand_dark,
    family = "Lato"
  ) +
  geom_text(

5. Offloading vs. Outsourcing: Ein Spektrum

Nicht jede KI-Nutzung wirkt gleich auf das Lernen. Entscheidend ist, ob die lernrelevante kognitive Verarbeitung erhalten bleibt oder eliminiert wird.

spectrum_examples <- tibble(
  position = c(0.12, 0.30, 0.50, 0.75, 0.92),
  label = c(
    "Taschenrechner\nfür Arithmetik",

    "Frage an KI ohne\neigenen Versuch",
    "KI schreibt\nden Aufsatz"
  ),
  y_offset = c(0.72, 0.72, 0.76, 0.72, 0.72)
)

gradient_data <- tibble(

  mutate(fill_value = x)
